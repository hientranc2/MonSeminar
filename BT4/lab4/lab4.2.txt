import random

random.seed(0)

for i in range(10):
   a = random.randint(1, 12)
   b = random.randint(1, 12)

   question = "What is " + str(a) + " x " + str(b) + "? "

   try:
       answer = int(input(question))

       if answer == a * b:
           print("Well done!")
       else:
           print("No.")
   except ValueError:
       print("Invalid input")
